<script>
const noMessages = [
"Are you sure, My beloved Kuu?",
"Think again!",
"You cannot click No!",
"Just click Yes already :P",
"Nice try, but No is not an option!"
];
let clickCount = 0;

function moveNo() {
const noBtn = document.getElementById('noBtn');
const card = document.querySelector('.card');

const maxX = card.clientWidth - noBtn.clientWidth - 40;
const maxY = 100;

const randomX = Math.floor(Math.random() * maxX) - (maxX / 2);
const randomY = Math.floor(Math.random() * maxY) - 20;

noBtn.style.transform = `translate(${randomX}px, ${randomY}px)`;

if (clickCount < noMessages.length) {
noBtn.innerText = noMessages[clickCount];
clickCount++;
} else {
noBtn.innerText = "Okay fine, click Yes!";
}
}

function sayYes() {
document.getElementById('questionSection').style.display = 'none';
document.getElementById('hiddenMessage').style.display = 'block';

// Memutar lagu MP3 wave to earth - bad saat tombol Yes diklik
const music = document.getElementById('bgMusic');
music.play().catch(error => {
console.log("Audio autoplay restricted by browser:", error);
});
}
